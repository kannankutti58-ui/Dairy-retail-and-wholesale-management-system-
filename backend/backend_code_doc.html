 22  export const verifyToken = (req, res, next) => {
 23    try {
 24      const authHeader = req.headers.authorization || req.headers['x-auth-token'];
 25      
 26      let token = '';
 27      if (authHeader && authHeader.startsWith('Bearer ')) {
 28        token = authHeader.split(' ')[1];
 29      } else if (authHeader) {
 30        token = authHeader;
 31      }
 32  
 33      if (!token) {
 34        return res.status(401).json({
 35          success: false,
 36          message: 'Access denied. No JWT token provided in Authorization header.'
 37        });
 38      }
 39  
 40      const decoded = jwt.verify(token, JWT_SECRET);
 41      req.user = decoded;
 42      next();
 43    } catch (error) {
 44      return res.status(401).json({
 45        success: false,
 46        message: 'Invalid or expired JWT token',
 47        error: error.message
 48      });
 49    }
 50  };
 51  
 52  export const requireRole = (...allowedRoles) => {
 53    return (req, res, next) => {
 54      if (!req.user) {
 55        return res.status(401).json({ success: false, message: 'Unauthorized. Please login.' });
 56      }
 57      if (!allowedRoles.includes(req.user.role) && req.user.role !== 'admin' && req.user.role !== 'proprietor') {
 58        return res.status(403).json({
 59          success: false,
 60          message: `Forbidden: Access requires one of roles: [${allowedRoles.join(', ')}]. Current role: ${req.user.role}`
 61        });
 62      }
 63      next();
 64    };
 65  };

193    calculateDistanceKm(lat1, lon1, lat2, lon2) {
194      const R = 6371; // Earth radius in km
195      const dLat = (lat2 - lat1) * (Math.PI / 180);
196      const dLon = (lon2 - lon1) * (Math.PI / 180);
197      const a =
198        Math.sin(dLat / 2) * Math.sin(dLat / 2) +
199        Math.cos(lat1 * (Math.PI / 180)) * Math.cos(lat2 * (Math.PI / 180)) *
200        Math.sin(dLon / 2) * Math.sin(dLon / 2);
201      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
202      return R * c;
203    }
